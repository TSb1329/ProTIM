    Dim basisPfad As String
    Dim dateien As Variant
    Dim i As Long
    
    basisPfad = "C:\Users\Thomas\OneDrive\EDV\Projekte\i4next\ProTIM - Test\src\"

dateien = Array( _
    "Forms\F_ProjektBearbeitung.txt" _

    For i = LBound(dateien) To UBound(dateien)
        ImportEineTextdatei basisPfad & dateien(i)
    Next i
    
    Hinweis "Import abgeschlossen."
End Sub

Private Sub ImportEineTextdatei(ByVal vollPfad As String)
    Dim objTyp As AcObjectType
    Dim objName As String
    Dim relPfad As String
    
    relPfad = Mid(vollPfad, InStr(1, vollPfad, "\src\") + 5)
    objName = Mid(vollPfad, InStrRev(vollPfad, "\") + 1)
    objName = Left(objName, Len(objName) - 4)
    
    If Left(relPfad, 6) = "Forms\" Then
        objTyp = acForm
    ElseIf Left(relPfad, 8) = "Queries\" Then
        objTyp = acQuery
    ElseIf Left(relPfad, 8) = "Reports\" Then
        objTyp = acReport
    ElseIf Left(relPfad, 8) = "Modules\" Then
        objTyp = acModule
    Else
        Err.Raise vbObjectError + 1000, , "Unbekannter Objekttyp: " & vollPfad
    End If
    
    On Error Resume Next
    DoCmd.Close objTyp, objName, acSaveNo
    On Error GoTo 0
    
    Application.LoadFromText objTyp, objName, vollPfad
End Sub
